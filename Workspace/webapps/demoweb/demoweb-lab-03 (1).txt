def insert_titanic_data(titanic_data):

    cursor = None
    conn = None
    try:
        # 1. 연결
        conn = pymysql.connect(host='localhost',
                               port=3306,
                               database='demoweb',
                               user='humanda6',
                               password='humanda6')
        
        # 2. 명령객체 만들기
        cursor = conn.cursor()
        
        # 3. 명령 실행 (SQL)
        sql = "insert into titanic values (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)"
        # cursor.execute(sql, titanic_data[0]) # 한 건의 데이터 처리
        cursor.executemany(sql, titanic_data) # 여러 건의 데이터 처리
        conn.commit() # 수정 내용을 원본 테이블에 일괄 적용

        # 4. 결과 처리 (SELECT 인 경우)
        
    except Exception as e:
        print('타이타닉 데이터 삽입 실패 : ', e)
        if conn:
            conn.rollback() # 수정 내용을 일괄 취소
    finally:
        # 5. 연결 종료
        if cursor:
            cursor.close()
        if conn:
            conn.close()

@data_bp.route('/init-titanic-table', methods=['GET'])
def init_titanic_table():

    data_util.create_titanic_table()

    # 파일에서 데이터 읽기
    bp_path = data_bp.root_path # 현재 blueprint의 경로
    root_path = Path(bp_path).parent #  부모 경로 
    file_path = os.path.join(root_path, 'data_files', 'titanic-train.csv')    
    df = pd.read_csv(file_path)

    df['Age'] = df['Age'].fillna(df['Age'].mean()) # db에 데이터 저장할 때 nan을 사용할 수 없기 때문에 전처리
    df['Cabin'] = df['Cabin'].fillna('') # db에 데이터 저장할 때 nan을 사용할 수 없기 때문에 전처리
    df['Embarked'] = df['Embarked'].fillna('') # db에 데이터 저장할 때 nan을 사용할 수 없기 때문에 전처리

    data_util.insert_titanic_data(df.values.tolist())

    return redirect(url_for('main.index'))
